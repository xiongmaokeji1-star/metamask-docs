pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/extensions/ERC20Pausable.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract USDTAdvanced is ERC20Pausable, Ownable {

    mapping(address => bool) public blacklist;
    uint256 public maxSupply;

    event Blacklisted(address indexed user);
    event UnBlacklisted(address indexed user);
    event DestroyedBlackFunds(address indexed user, uint256 balance);

    constructor(
        uint256 initialSupply_,
        uint256 maxSupply_
    ) ERC20("cyusd", "USDT") {

        require(maxSupply_ >= initialSupply_, "Max < Initial");

        uint256 supplyWithDecimals = initialSupply_ * 10 ** decimals();
        uint256 maxWithDecimals = maxSupply_ * 10 ** decimals();

        _mint(msg.sender, supplyWithDecimals);
        maxSupply = maxWithDecimals;
    }

    function mint(address to, uint256 amount) external onlyOwner {
        require(totalSupply() + amount <= maxSupply, "Exceeds max supply");
        _mint(to, amount);
    }

    function pause() external onlyOwner {
        _pause();
    }

    function unpause() external onlyOwner {
        _unpause();
    }

    function addBlacklist(address user) external onlyOwner {
        blacklist[user] = true;
        emit Blacklisted(user);
    }

    function removeBlacklist(address user) external onlyOwner {
        blacklist[user] = false;
        emit UnBlacklisted(user);
    }

    function destroyBlackFunds(address user) external onlyOwner {
        require(blacklist[user], "Not blacklisted");
        uint256 balance = balanceOf(user);
        _burn(user, balance);
        emit DestroyedBlackFunds(user, balance);
    }

    function _beforeTokenTransfer(
        address from,
        address to,
        uint256 amount
    ) internal override {

        require(!blacklist[from], "Sender blacklisted");
        require(!blacklist[to], "Receiver blacklisted");

        super._beforeTokenTransfer(from, to, amount);
    }
}
